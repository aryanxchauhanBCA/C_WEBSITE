2. Create a file and count the number of characters in it.

Code 
#include < stdio.h >
#include < dos.h > 
#include < conio.h >

 void main()
{
    FILE *fp;
    char file1[10];
    int data, count=0;

    clrscr();

    printf("Enter File Name: ");
    scanf("%s", file1);

    printf("Enter Data: \n");

    fp = fopen(file1, "w");

    if(fp == NULL)
    {
        printf("Error! File Cannot Open..");
    }
    else
    {
        do
        {
            data = getch();

            if(data != 19)
            {
                if(data == 13)
                {
                    printf("\n");
                    fputc('\n', fp);
                }
                else
                {
                    printf("%c", data);
                    fprintf(fp, "%c", data);
                }
            }
        }
        while(data != 19);

        fclose(fp);
    }

    fp = fopen(file1, "r");

    if(fp == NULL)
    {
        printf("Error In Opening File");
    }
    else
    {
        while((fgetc(fp)) != EOF)       //EOF means end of file = -1
        {
            count++;
        }

        fclose(fp);

        printf("\n\nTotal Number Of Characters In File = %d", count);
    }

    delay(3000);
}

3. Write a program to create a file and count the number of words in the file

Code 
#include < stdio.h >
#include < dos.h > 
#include < conio.h >

 void main()
{
    FILE *fp;
    char file1[10];
    int data, count=0, inword=0;

    clrscr();

    printf("Enter File Name: ");
    scanf("%s", file1);

    printf("Enter Data: \n");

    fp = fopen(file1, "w");

    if(fp == NULL)
    {
        printf("Error! File Cannot Open..");
    }
    else
    {
        do
        {
            data = getch();

            if(data != 19)
            {
                if(data == 13)
                {
                    printf("\n");
                    fputc('\n', fp);
                }
                else
                {
                    printf("%c", data);
                    fprintf(fp, "%c", data);
                }
            }
        }
        while(data != 19);

        fclose(fp);
    }

    fp = fopen(file1, "r");

    if(fp == NULL)
    {
        printf("Error In Opening File");
    }
    else
    {
        while((data = fgetc(fp)) != EOF)
        {
            if(data == ' ' || data == '\n' || data == '\t')
            {
                inword = 0;
            }
            else if(inword == 0)
            {
                count++;
                inword = 1;
            }
        }

        fclose(fp);

        printf("\n\nTotal Number Of Words In File = %d", count);
    }

    delay(3000);
}
