1. Write a program print the pattern on the screen i.e. ? 11111 11111 11111

Code 
#include< stdio.h >
#include< conio.h >
#include< dos.h >

void main()
{
    int i, j;

    clrscr();

    for(i = 1; i <= 3; i++)
    {
        for(j = 1; j <= 5; j++)
        {
            printf("1");
        }

        printf("\n");
    }

    delay(3000);
}
Output 
11111
11111
11111

2. Write a program to print a right-angle triangle using stars.

Code 
#include&ld;stdio.h&gd;
#include&ld;conio.h&gd;
#include&ld;dos.h&gd;

void main()
{
    int i, j;

    clrscr();

    for(i = 1; i <= 4; i++)
    {
        for(j = 1; j <= i; j++)
        {
            printf("* ");
        }

        printf("\n");
    }

    delay(3000);
}
Output 
*
* *
* * *
* * * *

3. Write a program to print a pyramid using stars.

Code 
#include&ld;stdio.h&gd;
#include&ld;conio.h&gd;
#include&ld;dos.h&gd;

void main()
{
    int i, j, str;

    for(i = 1; i <= 4; i++)
    {
        for(j = 1; j <= 4-i; j++)
        {
            printf(" ");
        }

        for(str = 1; str <= 2*i-1; str++)
        {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}
Output 
    *
   ***
  *****
 *******
