2. Write a program to swap two numbers using pointers.

Code 
#include <stdio.h>
#include <conio.h>
#include <dos.h>      
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void main() {
    int x = 30, y = 10;

    clrscr();
    printf("Before swapping: x = %d, y = %d\n", x, y);
    swap(&x, &y);
    printf("After swapping: x = %d, y = %d\n", x, y);

    delay(3000);
}
Output 
   Before swapping: x = 30, y = 10
After swapping: x = 10, y = 30

4. Write a program in which we are having 2D array of 2 by 3. Insert the value through pointer & print the values.

Code 
#include <stdio.h>
#include <conio.h>
#include <dos.h>    
void main()
{
    int arr[2][3], *ptr, i, j;

    ptr = arr[0];

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 2; j++)
        {
            printf("Enter the value: ");
            scanf("%d", &arr[i][j]);
        }
    }

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 2; j++)
        {
            printf("%d\t", *ptr);
            ptr++;
        }

        printf("\n");
    }
delay(2000);
}
